cmake_minimum_required(VERSION 3.25)
project(TRAMA VERSION 0.1.0 LANGUAGES CXX)
option(TRAMA_BUILD_TESTS "Build tests" ON)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
  find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
endif()
add_library(trama_core src/trama.cpp)
target_include_directories(trama_core PUBLIC include third_party/cpp-httplib ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(trama_core PUBLIC SQLite3::SQLite3 OpenSSL::Crypto Threads::Threads)
target_compile_definitions(trama_core PUBLIC CPPHTTPLIB_THREAD_POOL_COUNT=8)
target_compile_options(trama_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(trama apps/trama-cli/main.cpp)
target_link_libraries(trama PRIVATE trama_core)
add_executable(trama-server apps/trama-server/main.cpp)
target_link_libraries(trama-server PRIVATE trama_core)
if(TRAMA_BUILD_TESTS)
  enable_testing()
  add_executable(trama-tests tests/integration/tests.cpp)
  target_link_libraries(trama-tests PRIVATE trama_core)
  add_test(NAME trama-integration COMMAND trama-tests ${CMAKE_SOURCE_DIR})
endif()
